# TRUEWATCH edge service — Phase 0 scaffolding.
# Versions are pinned as each phase introduces the dependency it needs.
# Nothing here is installed by `npm run setup`; the edge service has its own
# virtual environment (see edge/README.md).

# --- Service (Phase 1) ---
fastapi
uvicorn[standard]
pydantic
python-dotenv
httpx

# --- Decode and classical vision (Phases 1, 3) ---
opencv-python-headless
numpy

# --- Appearance channel (Phase 2) ---
# Inference runs on the exported ONNX model, so onnxruntime is the runtime
# dependency. The training framework is a TRAINING dependency and is declared in
# training/, not here — see training/README.md for the licence note.
onnxruntime

# --- Tracking (Phase 3) ---
# ByteTrack is implemented in pipeline/tracking.py; scipy (BSD) supplies the Hungarian solver.
# Far-field tiling is implemented in pipeline/appearance.py, so no tiling package is needed.
scipy

# --- ANPR, face (Phase 5) ---
# PP-OCRv5 Devanagari and Latin recognition, served through paddleocr's
# TextRecognition (recognition-only: edge/anpr/detect_plate.py and rectify.py
# do plate localisation and line splitting with classical CV, not a second
# detector model). CPU-only, free, Apache-2.0. paddlepaddle is the runtime
# paddleocr needs to run; see edge/README.md for why this is the pretrained-
# serving path here instead of an ONNX export.
paddlepaddle
paddleocr
# cryptography.fernet seals the unblurred face crop at rest (edge/face/privacy.py).
cryptography
# YuNet (edge/face/detect.py) uses cv2.FaceDetectorYN — already covered by the
# opencv-python-headless pin above. Its weights are fetched at first use by
# edge/face/yunet_weights.py, the same pinned-URL/SHA-256 pattern as the
# appearance detector in edge/models/detector_weights.py.


# --- VLM explanation (Phase 6) + semantic footage search (Phase 6) ---
# transformers + torch serve both local Moondream 2 (edge/vlm/runtime.py) and
# SigLIP (edge/search/embed.py) on CPU. See edge/search/embed.py's module
# docstring for why plain transformers+torch was chosen here over an
# onnxruntime export of SigLIP's dual encoder: the call rate (once per
# keyframe at index time, once per typed query, once per confirmed alert for
# Moondream 2) never competes with the frame-rate-bound ONNX appearance
# model above.
#
# transformers is pinned below 5.0: Moondream 2's `trust_remote_code` model
# code (as of the pinned commit in edge/vlm/runtime.py) calls internal
# `PreTrainedModel` hooks that transformers 5.x renamed/removed
# (`_finalize_model_loading` -> `all_tied_weights_keys`, observed directly
# while building this module against transformers 5.17). 4.49.x is the
# version this repository was built and tested against; SigLIP loads fine on
# it too.
transformers>=4.49,<5
torch
einops        # Moondream 2's vision tower (moondream2/vision.py) imports it directly
pillow
sentencepiece

# --- Tests (Phase 4 onward) ---
pytest
