# TRUEWATCH edge service — Phase 0 scaffolding.
# Versions are pinned as each phase introduces the dependency it needs.
# Nothing here is installed by `npm run setup`; the edge service has its own
# virtual environment (see edge/README.md).

# --- Service (Phase 1) ---
fastapi
uvicorn[standard]
pydantic
python-dotenv
httpx

# --- Decode and classical vision (Phases 1, 3) ---
opencv-python-headless
numpy

# --- Appearance channel (Phase 2) ---
# Inference runs on the exported ONNX model, so onnxruntime is the runtime
# dependency. The training framework is a TRAINING dependency and is declared in
# training/, not here — see training/README.md for the licence note.
onnxruntime

# --- Tracking and tiling (Phases 5, 6) ---
# lap / scipy for ByteTrack association, sahi for tiled far-field inference.

# --- ANPR, face (Phase 5) ---
# PP-OCRv5 Devanagari and Latin recognition, served through paddleocr's
# TextRecognition (recognition-only: edge/anpr/detect_plate.py and rectify.py
# do plate localisation and line splitting with classical CV, not a second
# detector model). CPU-only, free, Apache-2.0. paddlepaddle is the runtime
# paddleocr needs to run; see edge/README.md for why this is the pretrained-
# serving path here instead of an ONNX export.
paddlepaddle
paddleocr
# cryptography.fernet seals the unblurred face crop at rest (edge/face/privacy.py).
cryptography
# YuNet (edge/face/detect.py) uses cv2.FaceDetectorYN — already covered by the
# opencv-python-headless pin above. Its weights are fetched at first use by
# edge/face/yunet_weights.py, the same pinned-URL/SHA-256 pattern as the
# appearance detector in edge/models/detector_weights.py.

# --- explanation, search (Phase 7) ---
# transformers + torch for Moondream 2 and SigLIP.

# --- Tests (Phase 4 onward) ---
pytest
